// TODAY: inline call
Order order = orders.save(request);

// the customer waits here
emailClient.sendConfirmation(order);

return ResponseEntity.status(201)
        .body(order);

// TOMORROW: publish a message
Order order = orders.save(request);

// the customer does not wait
publisher.orderCreated(order);

return ResponseEntity.status(201)
        .body(order);
